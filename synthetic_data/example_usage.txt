# Generate a new cohort through an OpenAI-compatible server (e.g. vLLM).
python -m synthetic_data.cli \
  --api-url http://localhost:8000/v1 \
  --model openai/gpt-oss-120b \
  --dataset-size 1000 \
  --num-features 2 \
  --num-confounders 1 \
  --num-effect-modifiers 1 \
  --structured-data \
  --clinical-question "Compare vinorelbine to gemcitabine for advanced/metastatic NSCLC." \
  --output-dir ./synthetic_output/my_dataset

# Continuous outcome
python -m synthetic_data.cli \
  --api-url http://localhost:8000/v1 \
  --outcome-type continuous --outcome-noise-std 1.0 \
  --num-features 10 --num-confounders 10 \
  --dataset-size 5000 \
  --output-dir ./synthetic_output/ten_confounders_continuous
